#contoh kode link redirect import crypto from 'node:crypto';

const KODE_TOKO = '716554639659'; const PRIVATE_KEY = process.env.QIOSPAY_PRIVATE_KEY;

function qiospayLink({ orderId, amount, redirectUrl = '', keterangan = '' }) { const signature = crypto .createHmac('sha256', PRIVATE_KEY) .update(${KODE_TOKO}|${orderId}|${amount}|${redirectUrl}) .digest('hex'); const q = new URLSearchParams({ order_id: orderId, amount: String(amount) }); if (redirectUrl) q.set('redirect_url', redirectUrl); if (keterangan) q.set('keterangan', keterangan); q.set('signature', signature); return https://my.qiospay.com/bayar/${KODE_TOKO}?${q}; }
